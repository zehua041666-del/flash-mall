package com.flashmall;

import org.mybatis.spring.annotation.MapperScan;
import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;

// scanBasePackages 必须给到 com.flashmall，
// 否则默认只扫启动类所在包（com.flashmall），
// user 模块里的 @Service / @RestController 不会被注册
@SpringBootApplication(scanBasePackages = "com.flashmall")
// 一次扫全部模块的 mapper 包，后续新增 Mapper 不用再单独标 @Mapper
@MapperScan("com.flashmall.**.mapper")
public class FlashMallApplication {
    public static void main(String[] args) {
        SpringApplication.run(FlashMallApplication.class, args);
    }
}

    `password`   VARCHAR(100) NOT NULL                COMMENT '密码(BCrypt)',
    `phone`      VARCHAR(20)  DEFAULT NULL            COMMENT '手机号',
    `email`      VARCHAR(100) DEFAULT NULL            COMMENT '邮箱',
    `nickname`   VARCHAR(50)  DEFAULT NULL            COMMENT '昵称',

    `status`     TINYINT      NOT NULL DEFAULT 1      COMMENT '1正常/0禁用',

package com.flashmall.cart.controller;

import com.flashmall.cart.dto.AddCartRequest;
import com.flashmall.cart.dto.CartItemResponse;
import com.flashmall.cart.service.CartService;
import com.flashmall.common.result.Result;
import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@RequestMapping("/cart")
@RequiredArgsConstructor
public class CartController {

    private final CartService cartService;

    /**
     * 加入购物车
     * POST /cart/add
     */
    @PostMapping("/add")
    public Result<Void> addCart(@Valid @RequestBody AddCartRequest request) {
        cartService.addCart(request);
        return Result.ok(null);
    }

    /**
     * 获取当前用户的购物车列表
     * GET /cart/list
     */
    @GetMapping("/list")
    public Result<List<CartItemResponse>> getCartList() {
        return Result.ok(cartService.getCartList());
    }

    /**
     * 修改购物车商品数量
     * PUT /cart/quantity?skuId=1&quantity=5
     */
    @PutMapping("/quantity")
    public Result<Void> updateQuantity(
            @RequestParam Long skuId,
            @RequestParam Integer quantity) {
        cartService.updateQuantity(skuId, quantity);
        return Result.ok(null);
    }

    /**
     * 删除购物车商品
     * DELETE /cart/1
     */
    @DeleteMapping("/{skuId}")
    public Result<Void> removeCartItem(@PathVariable Long skuId) {
        cartService.removeCartItem(skuId);
        return Result.ok(null);
    }
}

package com.flashmall.cart.service.impl;

import com.flashmall.cart.dto.AddCartRequest;
import com.flashmall.cart.dto.CartItemResponse;
import com.flashmall.cart.service.CartService;
import com.flashmall.common.context.UserContext;
import com.flashmall.product.entity.ProductSku;
import com.flashmall.product.service.ProductService;
import lombok.RequiredArgsConstructor;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.stereotype.Service;

import java.math.BigDecimal;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import java.util.function.Function;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
public class CartServiceImpl implements CartService {

    private final StringRedisTemplate redisTemplate;
    private final ProductService productService;

    // Redis 购物车的 Key 前缀
    private static final String CART_KEY_PREFIX = "cart:";

    /**
     * 获取当前用户的 Redis Key，例如 cart:1001
     */
    private String getCartKey() {
        Long userId = UserContext.getUserId();
        // 如果拦截器没拦住，userId 为空，抛异常
        if (userId == null) {
            throw new RuntimeException("用户未登录");
        }
        return CART_KEY_PREFIX + userId;
    }

    @Override
    public void addCart(AddCartRequest request) {
        String key = getCartKey();
        String field = String.valueOf(request.getSkuId());
        
        // opsForHash().increment 是原子的。如果该 field 不存在，会初始化为 0 再累加。
        // 比如原本买了 2 个，再加 3 个，Redis 会直接变成 5。
        redisTemplate.opsForHash().increment(key, field, request.getQuantity());
    }

    @Override
    public List<CartItemResponse> getCartList() {
        String key = getCartKey();
        
        // 1. 从 Redis 获取所有购物车项（key 是 skuId，value 是数量）
        Map<Object, Object> entries = redisTemplate.opsForHash().entries(key);
        if (entries.isEmpty()) {
            return new ArrayList<>();
        }

        // 2. 提取所有的 skuId 并转为 Long 集合
        List<Long> skuIds = entries.keySet().stream()
                .map(obj -> Long.valueOf(obj.toString())) 
                .toList();

        // 3. 跨模块调用：通过商品模块的接口拿数据，绝对不查数据库
        List<ProductSku> skuList = productService.getSkuByIds(skuIds);
        
        // 将查出来的商品信息转为 Map，方便根据 skuId 快速匹配
        Map<Long, ProductSku> skuMap = skuList.stream()
                .collect(Collectors.toMap(ProductSku::getId, Function.identity()));

        // 4. 组装返回给前端的列表
        List<CartItemResponse> responseList = new ArrayList<>();
        for (Map.Entry<Object, Object> entry : entries.entrySet()) {
            Long skuId = Long.valueOf(entry.getKey().toString());
            Integer quantity = Integer.valueOf(entry.getValue().toString());

            ProductSku sku = skuMap.get(skuId);
            if (sku == null) {
                // 如果商品被下架删除了（skuMap 里找不到），可以选择跳过或者当垃圾数据清理
                continue; 
            }

            CartItemResponse item = new CartItemResponse();
            item.setSkuId(skuId);
            item.setSkuName(sku.getName());
            item.setSkuImage(sku.getImage());
            item.setPrice(sku.getPrice());
            item.setQuantity(quantity);
            
            // 计算单行小计：单价 * 数量
            item.setTotalPrice(sku.getPrice().multiply(new BigDecimal(quantity)));

            responseList.add(item);
        }

        return responseList;
    }

    @Override
    public void updateQuantity(Long skuId, Integer quantity) {
        String key = getCartKey();
        String field = String.valueOf(skuId);
        
        if (quantity <= 0) {
            // 如果数量减到 0，直接删除该项
            redisTemplate.opsForHash().delete(key, field);
        } else {
            // 直接覆盖更新数量
            redisTemplate.opsForHash().put(key, field, String.valueOf(quantity));
        }
    }

    @Override
    public void removeCartItem(Long skuId) {
        String key = getCartKey();
        redisTemplate.opsForHash().delete(key, String.valueOf(skuId));
    }
}

package com.flashmall.common.util;

import io.jsonwebtoken.Claims;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import javax.crypto.SecretKey;
import java.nio.charset.StandardCharsets;
import java.util.Date;

/**
 * JWT 工具类：生成 token、解析 token。
 * 放在 common 模块，因为 W7 拆服务后 gateway 也要校验 token，两边共用一份。
 */
@Component
public class JwtUtil {

    @Value("${jwt.secret}")
    private String secret;

    @Value("${jwt.expiration}")
    private long expiration;

    /**
     * 根据密钥字符串生成 SecretKey。
     * jjwt 0.12.x 要求密钥至少 256 位，否则抛 WeakKeyException。
     */
    private SecretKey getKey() {
        return Keys.hmacShaKeyFor(secret.getBytes(StandardCharsets.UTF_8));
    }

    /**
     * 生成 token，payload 里放 userId 和 username。
     */
    public String generateToken(Long userId, String username) {
        Date now = new Date();
        Date expiry = new Date(now.getTime() + expiration);

        return Jwts.builder()
                .subject(String.valueOf(userId))      // subject 存 userId
                .claim("username", username)          // 额外 claim 存 username
                .issuedAt(now)
                .expiration(expiry)
                .signWith(getKey())
                .compact();
    }

    /**
     * 解析 token，返回 Claims（payload）。
     * token 无效 / 过期 → 抛 JwtException，调用方自己 catch。
     */
    public Claims parseToken(String token) {
        return Jwts.parser()
                .verifyWith(getKey())
                .build()
                .parseSignedClaims(token)
                .getPayload();
    }

    /**
     * 从 token 里取 userId。
     */
    public Long getUserId(String token) {
        return Long.valueOf(parseToken(token).getSubject());
    }

    /**
     * 从 token 里取 username。
     */
    public String getUsername(String token) {
        return parseToken(token).get("username", String.class);
    }
}

package com.flashmall.product.service.impl;

import java.math.BigDecimal;
import java.util.Collections;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

import org.springframework.stereotype.Service;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
import com.flashmall.common.result.PageResult;
import com.flashmall.product.dto.response.ProductListResponse;
import com.flashmall.product.entity.ProductSku;
import com.flashmall.product.entity.ProductSpu;
import com.flashmall.product.mapper.ProductSkuMapper;
import com.flashmall.product.mapper.ProductSpuMapper;
import com.flashmall.product.service.ProductService;

import lombok.RequiredArgsConstructor;

@Service
@RequiredArgsConstructor
public class ProductServiceImpl implements ProductService {
    private final ProductSpuMapper productSpuMapper;
    private final ProductSkuMapper productSkuMapper;

    @Override
    public PageResult<ProductListResponse> listProducts(int pageNum, int pageSize) {
        Page<ProductSpu> page = productSpuMapper.selectPage(new Page<>(pageNum, pageSize),
                new LambdaQueryWrapper<ProductSpu>().eq(ProductSpu::getStatus, 1).orderByDesc(ProductSpu::getId));
        List<ProductSpu> spuList = page.getRecords();
        if (spuList.isEmpty()) {
            return new PageResult<>(page.getTotal(), page.getCurrent(), page.getSize(), Collections.emptyList());
        }
        List<Long> spuIds = spuList.stream().map(ProductSpu::getId).toList();
        List<ProductSku> allSkus = productSkuMapper.selectList(
                new LambdaQueryWrapper<ProductSku>().in(ProductSku::getSpuId, spuIds).eq(ProductSku::getStatus, 1));
        Map<Long, List<ProductSku>> skusBySpu = allSkus.stream().collect(Collectors.groupingBy(ProductSku::getSpuId));
        List<ProductListResponse> list = spuList.stream().map(spu -> {
            ProductListResponse item = new ProductListResponse();
            item.setSpuId(spu.getId());
            item.setName(spu.getName());
            item.setMainImage(spu.getMainImage());

            List<ProductSku> skus = skusBySpu.getOrDefault(spu.getId(), Collections.emptyList());
            BigDecimal minPrice = skus.stream().map(ProductSku::getPrice).min(BigDecimal::compareTo)
                    .orElse(BigDecimal.ZERO);
            item.setMinPrice(minPrice);
            return item;
        }).toList();
        return new PageResult<>(page.getTotal(), page.getCurrent(), page.getSize(), list);
    }

    @Override
    public List<ProductSku> getSkuByIds(List<Long> skuIds) {
        if (skuIds == null || skuIds.isEmpty()) {
            return Collections.emptyList();
        }
        return productSkuMapper.selectList(
                new LambdaQueryWrapper<ProductSku>().in(ProductSku::getId, skuIds));
    }

}

package com.flashmall.user.interceptor;

import com.flashmall.common.context.UserContext;
import com.flashmall.common.exception.BizException;
import com.flashmall.common.util.JwtUtil;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Component;
import org.springframework.util.StringUtils;
import org.springframework.web.servlet.HandlerInterceptor;

@Component
@RequiredArgsConstructor
public class JwtInterceptor implements HandlerInterceptor {

    private final JwtUtil jwtUtil;

    @Override
    public boolean preHandle(HttpServletRequest request,HttpServletResponse response ,Object handler) throws Exception{
        String authHeader =request.getHeader("Authorization");

        if (!StringUtils.hasText(authHeader)|| !authHeader.startsWith("Bearer ")) {
            throw new BizException(401,"未登录或token已过期");
        }
        String token = authHeader.substring(7);
        try{
            Long userId =jwtUtil.getUserId(token);
            UserContext.setUserId(userId);
            return true;
        }
        catch (Exception e){
            throw new BizException(401,"Token无效或已过期");
        }

    }
    @Override
    public void afterCompletion(HttpServletRequest request, HttpServletResponse response, Object handler, Exception ex) throws Exception {
        // 7. 请求结束，必须清理 ThreadLocal，防止内存泄漏
        UserContext.clear();
    }
}

package com.flashmall.user.service.impl;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.flashmall.common.exception.BizException;
import com.flashmall.common.util.JwtUtil;
import com.flashmall.user.dto.request.LoginRequest;
import com.flashmall.user.dto.request.RegisterRequest;
import com.flashmall.user.dto.response.LoginResponse;
import com.flashmall.user.entity.User;
import com.flashmall.user.mapper.UserMapper;
import com.flashmall.user.service.UserService;
import lombok.RequiredArgsConstructor;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;

@Service
@RequiredArgsConstructor
public class UserServiceImpl implements UserService {

    private final UserMapper userMapper;
    private final PasswordEncoder passwordEncoder;
    private final JwtUtil jwtUtil;

    @Override
    public void register(RegisterRequest request) {
        // 先查用户名是否已存在
        // MyBatis-Plus 逻辑删除：这里自动带 WHERE deleted = 0，已删的用户名可以复用
        //这里是MP的实际应用 我现在不太清楚 所以 现在我先用我的理解来描述一下 这个的过程 首先就是 他是直接先看request.getusername
        //然后 后面是
        Long count = userMapper.selectCount(
                new LambdaQueryWrapper<User>().eq(User::getUsername, request.getUsername())
        );
        if (count > 0) {
            throw new BizException(400, "用户名已存在");
        }

        User user = new User();
        user.setUsername(request.getUsername());
        // BCrypt 加密，绝不能明文入库
        user.setPassword(passwordEncoder.encode(request.getPassword()));
        user.setEmail(request.getEmail());
        user.setStatus(1);   // 默认正常

        userMapper.insert(user);
    }

    @Override
    public LoginResponse login(LoginRequest request) {
        User user = userMapper.selectOne(
                new LambdaQueryWrapper<User>().eq(User::getUsername, request.getUsername())
        );

        // 用户不存在 和 密码错误，故意返回同一个提示
        // 防止攻击者通过错误信息判断"用户名是否存在"（用户名枚举攻击）
        if (user == null || !passwordEncoder.matches(request.getPassword(), user.getPassword())) {
            throw new BizException(400, "用户名或密码错误");
        }

        if (user.getStatus() == null || user.getStatus() != 1) {
            throw new BizException(403, "账号已被禁用");
        }

        String token = jwtUtil.generateToken(user.getId(), user.getUsername());
        return new LoginResponse(token, user.getId(), user.getUsername(), user.getNickname());
    }
}

import axios from 'axios'
import { ElMessage } from 'element-plus'
import { getToken, clearToken } from '@/utils/token'
import router from '@/router'

const request = axios.create({
  baseURL: '/api',
  timeout: 10000
})

// 请求拦截：统一注入 token，业务代码不用每次手动加 header
request.interceptors.request.use(
  (config) => {
    const token = getToken()
    if (token) {
      config.headers.Authorization = `Bearer ${token}`
    }
    return config
  },
  (error) => Promise.reject(error)
)

// 响应拦截：与后端 Result{code, message, data} 约定对齐
// code=0 成功直接吐 data，业务代码拿到的就是纯数据
request.interceptors.response.use(
  (response) => {
    const res = response.data
    if (res.code !== 0) {
      ElMessage.error(res.message || '请求失败')
      if (res.code === 401) {
        clearToken()
        router.push('/login')
      }
      return Promise.reject(new Error(res.message || 'Error'))
    }
    return res.data
  },
  (error) => {
    // HTTP 层异常（网络断、超时、500 未走到 Result 包装）
    ElMessage.error(error.message || '网络异常')
    return Promise.reject(error)
  }
)

export default request

import { ref, onMounted, computed } from 'vue'
import { ElMessage } from 'element-plus'
import { getCartList, updateCartQuantity, removeCartItem } from '@/api/cart'

// 状态变量
const cartList = ref([])
const loading = ref(false)

// 自动计算所有商品的小计总和
const totalAmount = computed(() => {
  let sum = 0
  cartList.value.forEach(item => {
    sum += Number(item.totalPrice)
  })
  return sum.toFixed(2) // 保留两位小数
})

// 获取购物车列表
async function fetchData() {
  loading.value = true
  try {
    const res = await getCartList()
    // res 直接就是后端返回的 List
    cartList.value = res || []
  } catch (e) {
    // 错误已在 request.js 拦截器处理
  } finally {
    loading.value = false
  }
}

// 修改数量 (el-input-number 触发)
async function handleQuantityChange(skuId, newQuantity) {
  try {
    await updateCartQuantity(skuId, newQuantity)
    ElMessage.success('数量已更新')
    fetchData() // 重新拉取最新数据，保证总价实时刷新
  } catch (e) {
    // 如果失败，重新拉一次数据，把页面上的错误数字还原
    fetchData()
  }
}

// 删除商品
async function handleDelete(skuId) {
  try {
    await removeCartItem(skuId)
    ElMessage.success('删除成功')
    fetchData() // 重新拉取数据刷新页面
  } catch (e) {
    //
  }
}

// 页面加载时请求数据
onMounted(() => {
  fetchData()
})
</script>

            @keyup.enter="handleLogin" />
        </el-form-item>
        <el-button type="primary" style="width: 100%" :loading="loading" @click="handleLogin">
          登录
        </el-button>

<script setup>
import { reactive, ref } from 'vue'
import { useRouter } from 'vue-router'
import { ElMessage } from 'element-plus'
import { login } from '@/api/user'
import { useUserStore } from '@/stores/user'

const router = useRouter()
const userStore = useUserStore()

const form = reactive({
  username: '',
  password: ''
})

// loading 防止用户连点导致重复请求
const loading = ref(false)
const passwordInputRef = ref(null)

async function handleLogin() {
  if (loading.value) return

  if (!form.username || !form.password) {
    ElMessage.warning('请输入用户名和密码')
    return
  }

  loading.value = true
  try {
    // 调用登录接口，request.js 拦截器已经把 Result 解包成 data
    // 所以 res 直接就是 { token, userId, username, nickname }
    const res = await login({
      username: form.username,
      password: form.password
    })

    // 控制台打印 JWT，验收标准要求
    console.log('JWT:', res.token)

    // 存进 Pinia + localStorage
    userStore.loginSuccess(res.token, {
      userId: res.userId,
      username: res.username,
      nickname: res.nickname
    })

    ElMessage.success('登录成功')
    router.push('/')
  } catch (e) {
    // 错误提示已经在 request.js 拦截器里弹过了，这里不用再弹
  } finally {
    loading.value = false
  }
}
</script>
